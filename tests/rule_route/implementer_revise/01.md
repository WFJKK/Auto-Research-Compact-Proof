```python
"""test_bound: the margin of the leaf's first input, claimed for the whole leaf (unsound on purpose)."""

from fractions import Fraction

import numpy as np

from core.check.evaluate import CheckError
from core.check.general import GeneralProgram

NAME = "test_bound"
CATEGORY = "test_bound"
APPLIES_TO = "general"
DESCRIPTION = 'Leaf rule {"leaf": "test_bound"}. Checks one input and claims its margin for the leaf.'


def check_leaf(prog, ranges, label, counter):
    if not isinstance(prog, GeneralProgram):
        raise CheckError("test_bound is only for networks beyond linear maps")
    tokens = np.array([[lo for lo, _ in ranges]], dtype=np.int64)
    ok, margins = prog.run_singles(tokens, np.array([label], dtype=np.int64), counter)
    m = Fraction(float(margins[0]))
    return m > 0, m, -1
```
